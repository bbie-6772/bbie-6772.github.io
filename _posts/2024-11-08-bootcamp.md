#### 3진법 뒤집기

```jsx
function solution(n) {
    let answer = 0;
    let trinary = [];
    
    trinary = n.toString(3)
    
    for (let i = trinary.length - 1;i >= 0;i--) {
        answer += trinary[i];
    }
    
    answer = parseInt(answer, 3)
    
    return answer;
}
```
