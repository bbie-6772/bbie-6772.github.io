#### 두 수의 차 / 곱 / 몫(나눗셈) / 합

```jsx
// 두 수의 차
function solution1(num1, num2) {
    let answer = 0;
    answer = num1 - num2;
    return answer;
}
// 두 수의 곱
function solution2(num1, num2) {
    let answer = 0;
    answer = num1 * num2
    return answer;
}
// 나눗셈
function solution(num1, num2) {
    let answer = 0;
    // 내림을 통해 나눠진 정수만을 표시
    answer = Math.floor(num1 / num2);
    return answer;
}
// 합
function solution(num1, num2) {
    let answer = 0;
    answer = num1 + num2;
    return answer;
}
```

#### 각도기

- 조건  
각에서 0도 초과 90도 미만은 예각, 90도는 직각, 90도 초과 180도 미만은 둔각 180도는 평각으로 분류합니다. 각 angle이 매개변수로 주어질 때 예각일 때 1, 직각일 때 2, 둔각일 때 3, 평각일 때 4를 return하도록 solution 함수를 완성해주세요.

- 제한사항  
    1. 0 < angle ≤ 180  
    2. angle은 정수입니다.

```jsx
function solution(angle) {
    let answer = 0;
    
    if (angle < 90) {
        answer = 1;
    } else if (angle === 90) {
        answer = 2;
    } else if (angle < 180) {
        answer = 3;
    //제한 사항의 범위안에선 굳이 추가연산없이 else를 써도 된다
    } else {
        answer = 4;
    }
    return answer;
}
```

#### 짝수의 합

- 조건  
정수 n이 주어질 때, n이하의 짝수를 모두 더한 값을 return 하도록 solution 함수를 작성해주세요.

- 제한사항  
    1. 0 < n ≤ 1000

```jsx
function solution(n) {
    let answer = 0;
    for (let i = 0;i <= n;i++){
        // 짝수 = 2로 나눠지는 수 이므로
        if (i % 2 === 0) {
            answer += i;
        }
    }
    return answer;
}
```

#### 배열의 평균값 / 평균 구하기

- 조건  
정수 배열 numbers가 매개변수로 주어집니다. numbers의 원소의 평균값을 return하도록 solution 함수를 완성해주세요.

- 제한사항  
    1. 0 ≤ numbers의 원소 ≤ 1,000  
    2. 1 ≤ numbers의 길이 ≤ 100  
    3. 정답의 소수 부분이 .0 또는 .5인 경우만 입력으로 주어집니다.

```jsx
// 배열의 평균값 
function solution(numbers) {
    let answer = 0;
    let sum = 0;
    
    numbers.forEach((val, idx) => {
        //마지막 수 확인
        if (idx + 1 === numbers.length) {
            sum += val;
            answer = sum / numbers.length;
        } else {
            sum += val;
        }
    });
    
    return answer;
}

// 평균 구하기 > 다른방법이 없나 찾다가 reduce 함수를 알게됨
function solution(arr) {
    let answer = arr.reduce((a, b) => a += b) / arr.length
    return answer;
}

//reduce 는 array의 각 값들에 주어진 함수를 실행하여 누적값을 반환함
// [] 는 옵션 매개변수를 뜻함
array.reduce(function (누적값, 현재값, [현재인덱스], [배열(array)]{
    함수 내용 
}))

// 화살표 함수 버전
array.reduce((누적값, 현재값, [현재인덱스], [배열(array)]) => {함수})
```

#### 3진법 뒤집기

- 조건  
자연수 n이 매개변수로 주어집니다. n을 3진법 상에서 앞뒤로 뒤집은 후, 이를 다시 10진법으로 표현한 수를 return 하도록 solution 함수를 완성해주세요.

- 제한사항  
    1. n은 1 이상 100,000,000 이하인 자연수입니다.

```jsx
function solution(n) {
    let answer = 0;
    let trinary = [];
    
    trinary = n.toString(3)
    
    for (let i = trinary.length - 1;i >= 0;i--) {
        answer += trinary[i];
    }
    answer = parseInt(answer, 3)
    return answer;
}

// toString 을 이용해 (진수) 로 변경
// paresInt를 이용해 (변경할 값,진수명) 10진수로 재변환
```

SELECT  ANIMAL_ID
        , NAME
        , SEX_UPON_INTAKE

ORDER BY 1
